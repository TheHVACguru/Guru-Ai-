    // Initialize the interface when page loads
    document.addEventListener('DOMContentLoaded', function() {
        initializeInterface();
        updateTimeDisplay();
        populateSystemGrid();
        
        // Update time every second to keep it current
        setInterval(updateTimeDisplay, 1000);
        
        // Simulate system activity with periodic updates
        setInterval(simulateSystemActivity, 2000);
    });

    // Update the time and date display in the reactor core
    function updateTimeDisplay() {
        const now = new Date();
        const timeElement = document.getElementById('currentTime');
        const dateElement = document.getElementById('currentDate');
        
        // Format time as HH:MM for that authentic tech interface look
        const timeString = now.getHours().toString().padStart(2, '0') + ':' + 
                          now.getMinutes().toString().padStart(2, '0');
        
        // Format date as MM/DD/YY to match the reference image
        const dateString = (now.getMonth() + 1).toString().padStart(2, '0') + '/' +
                          now.getDate().toString().padStart(2, '0') + '/' +
                          now.getFullYear().toString().substr(-2);
        
        timeElement.textContent = timeString;
        dateElement.textContent = dateString;
    }

    // Create the system grid with realistic technical labels
    function populateSystemGrid() {
        const systemGrid = document.querySelector('.system-grid');
        const gridLabels = [
            'PWR', 'SYS', 'NAV', 'COM', 'WPN', 'SHD',
            'THR', 'STB', 'GYR', 'ALT', 'VEL', 'TMP',
            'O2', 'CO2', 'H2O', 'FUL', 'BAT', 'GEN',
            'RAD', 'MAG', 'GRV', 'IRT', 'OPT', 'AUD'
        ];
        
        // Create each grid cell with a technical system abbreviation
        gridLabels.forEach(label => {
            const cell = document.createElement('div');
            cell.className = 'grid-cell';
            cell.textContent = label;
            
            // Add some cells that appear "active" with different styling
            if (Math.random() > 0.7) {
                cell.style.background = 'rgba(255, 153, 0, 0.2)';
                cell.style.color = '#ffcc00';
            }
            
            systemGrid.appendChild(cell);
        });
    }

    // Simulate realistic system activity by updating various interface elements
    function simulateSystemActivity() {
        // Randomly update some grid cells to show system changes
        const gridCells = document.querySelectorAll('.grid-cell');
        gridCells.forEach(cell => {
            if (Math.random() > 0.9) {
                // Briefly highlight cell to show activity
                cell.style.background = 'rgba(0, 255, 170, 0.4)';
                cell.style.boxShadow = '0 0 10px #00ffaa';
                
                setTimeout(() => {
                    cell.style.background = 'rgba(0, 170, 255, 0.1)';
                    cell.style.boxShadow = 'none';
                }, 500);
            }
        });
        
        // Update technical readouts with slight variations to show live data
        updateTechnicalReadouts();
    }

    // Update the technical readout values to simulate live system monitoring
    function updateTechnicalReadouts() {
        const readoutValues = document.querySelectorAll('.readout-value');
        const baseValues = ['2847°K', '12.7 GW', '97.3%', 'NOMINAL', 'STABLE', '8.4 T'];
        
        readoutValues.forEach((element, index) => {
            if (index < 3 && Math.random() > 0.8) {
                // Occasionally update numeric values with small variations
                const currentText = element.textContent;
                const variation = (Math.random() - 0.5) * 0.2;
                
                if (index === 0) { // Temperature
                    const temp = 2847 + Math.floor(variation * 10);
                    element.textContent = temp + '°K';
                } else if (index === 1) { // Power
                    const power = (12.7 + variation).toFixed(1);
                    element.textContent = power + ' GW';
                } else if (index === 2) { // Efficiency
                    const eff = (97.3 + variation).toFixed(1);
                    element.textContent = eff + '%';
                }
                
                // Reset to base value after brief change
                setTimeout(() => {
                    element.textContent = baseValues[index];
                }, 1500);
            }
        });
    }

        }, 2000);
    }

    // Display system messages (in a real implementation, this might be a notification system)
    function showSystemMessage(message) {
        // For this demo, we'll use a simple alert, but in production you'd want a proper notification overlay
        console.log('JARVIS SYSTEM:', message);

        const notification = document.createElement('div');
